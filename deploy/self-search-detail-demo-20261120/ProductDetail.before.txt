function previewAvailabilityLabel(
  availability: "in_stock" | "unknown" | undefined
) {
  return availability === "in_stock" ? "有货" : "可询价";
}

const fmt = (n: number | null | undefined) => (n ?? 0).toLocaleString("zh-CN");

interface LadderTier {
  minQty: number;
  price: number;
}

/* ── 阶梯价工具（与搜索页一致，兼容 {minQty,price} 与 {qty,price} 两种历史格式） ── */
function ladderOf(
  ladderPrices: unknown,
  priceEx: string | null,
  moq: number | null
): LadderTier[] {
  if (Array.isArray(ladderPrices) && ladderPrices.length > 0) {
    const tiers = (ladderPrices as Array<Record<string, unknown>>)
      .map(t => ({ minQty: Number(t.minQty ?? t.qty), price: Number(t.price) }))
      .filter(t => Number.isFinite(t.minQty) && Number.isFinite(t.price))
      .sort((a, b) => a.minQty - b.minQty);
    if (tiers.length > 0) return tiers;
  }
  return [{ minQty: moq || 1, price: Number(priceEx ?? 0) }];
}

function hasRealLadder(ladderPrices: unknown): boolean {
  return (
    Array.isArray(ladderPrices) &&
    (ladderPrices as Array<Record<string, unknown>>).some(
      t =>
        Number.isFinite(Number(t.minQty ?? t.qty)) &&
        Number.isFinite(Number(t.price))
    )
  );
}

function tierIndex(ladder: LadderTier[], qty: number) {
  let idx = 0;
  ladder.forEach((t, i) => {
    if (qty >= t.minQty) idx = i;
  });
  return idx;
}

/**
 * 商品详情页 /product/:id
 * 布局：导航栏 / 面包屑 / 主区（图片 + 规格参数 + 价格购买区）/ 详细参数表 / 相关商品推荐
 */
export default function ProductDetail() {
  const [, params] = useRoute("/product/:id");
  const [, navigate] = useLocation();
  const id = Number(params?.id);
  const validId = Number.isInteger(id) && id > 0;
  const { user, isAuthenticated, loading: authLoading } = useAuth();
  const { launchInquiry, inquiryEnterpriseChooser } = useEnterpriseInquiryLauncher();
  const [loginOpen, setLoginOpen] = useState(false);
  const utils = trpc.useUtils();

  const detailInput = useMemo(
    () => ({ id: validId ? id : 1, deferRelated: true }),
    [id, validId]
  );
  // 报价详情是登录配额接口。游客仅请求没有价格、数量或供应商字段的安全预览。
  const {
    data,
    isLoading: detailLoading,
    isError: detailError,
    error: detailQueryError,
  } = trpc.search.detail.useQuery(detailInput, {
    enabled: validId && !authLoading && isAuthenticated,
    trpc: { context: { skipBatch: true } },
    // detail 同样计入每日查询配额；详情页停留、回到窗口或重连不得自动重复查询。
    retry: false,
    refetchOnWindowFocus: false,
    refetchOnReconnect: false,
    meta: { expectedErrorCodes: ["NOT_FOUND"] },
  });
  const {
    data: previewDetail,
    isLoading: previewLoading,
    isError: previewError,
    error: previewQueryError,
  } = trpc.search.previewDetail.useQuery(detailInput, {
    enabled: validId && !authLoading && !isAuthenticated,
    trpc: { context: { skipBatch: true } },
    retry: (count, err) => err.data?.code !== "NOT_FOUND" && count < 2,
    meta: { expectedErrorCodes: ["NOT_FOUND"] },
  });
  const isLoading =
    authLoading || (isAuthenticated ? detailLoading : previewLoading);
  const isError = isAuthenticated ? detailError : previewError;
  const error = isAuthenticated ? detailQueryError : previewQueryError;

  const item = data?.item;
  // Deferred recommendations cannot hold the detail response or technical parameters open.
  const relatedToken = data?.relatedReadToken ?? null;
  const {
    data: deferredRelated,
    isFetching: relatedLoading,
    isError: relatedError,
  } = trpc.search.relatedDetail.useQuery(
    { id: validId ? id : 1, token: relatedToken ?? "" },
    {
      enabled: isAuthenticated && !!item && !!relatedToken,
      retry: false,
      refetchOnWindowFocus: false,
      refetchOnReconnect: false,
      staleTime: 120_000,
      trpc: { context: { skipBatch: true } },
    }
  );
  const related = deferredRelated?.related ?? data?.related ?? [];
  const mainExposureRef = useMaterialExposure<HTMLDivElement>(
    data?.exposureToken,
    String(item?.id ?? "")
  );
  const relatedExposureRef = useMaterialExposure<HTMLDivElement>(
    deferredRelated?.relatedExposureToken ?? data?.relatedExposureToken,
    related.map(row => row.id).join(",")
  );

  /* 后台数据字典参数 */
  const specsBatchInput = useMemo(
    () => ({ partNumbers: item?.partNumber ? [item.partNumber] : [] }),
    [item?.partNumber]
  );
  const { data: specsMap, isFetching: specsLoading } =
    trpc.material.getSpecsBatch.useQuery(specsBatchInput, {
      enabled:
        isAuthenticated &&
        Boolean(item?.partNumber && item.partNumber.length >= 3),
      staleTime: 120_000,
      retry: false,
      refetchOnWindowFocus: false,
      trpc: { context: { skipBatch: true } },
    });
  const specs = item?.partNumber ? specsMap?.[item.partNumber] : undefined;
  const specEntries = specs?.specs ? Object.entries(specs.specs) : [];

  const moq = item?.moq || 1;
  const ladder = useMemo(
    () => (item ? ladderOf(item.ladderPrices, item.priceEx, item.moq) : []),
    [item]
  );
  const [qty, setQty] = useState(1);
  const [quantityTouched, setQuantityTouched] = useState(false);
  useEffect(() => {
    if (item) {
      setQty(item.moq || 1);
      setQuantityTouched(false);
    }
  }, [item]);
  const ti = tierIndex(ladder, qty);
  const unitPrice = ladder[ti]?.price ?? Number(item?.priceEx ?? 0);
  const subtotal = +(unitPrice * qty).toFixed(2);
  const step = moq >= 100 ? 100 : moq >= 10 ? 10 : 1;
  const negotiable =
    !!item &&
    !(Number(item.priceEx ?? 0) > 0) &&
    !(
      Array.isArray(item.ladderPrices) &&
      (item.ladderPrices as Array<Record<string, unknown>>).some(
        t => Number(t?.price) > 0
      )
    );
  const untaxed = !!item && item.taxIncluded === false && !negotiable;
  const displayedLeadTime = displayInventoryLeadTime(item?.leadTime);
  const quantityDisplay = item
    ? inventoryQuantityDisplay(item)
    : { spotQuantity: null, orderQuantity: null };
  // Order-only availability is quote-only; it never participates in cart, checkout, or pricing math.
  const canTransact = quantityDisplay.spotQuantity !== null;

  const photos = useMemo(() => (item ? productImageSources(item) : []), [item]);

  /*
   * 供应商关注：与搜索页共用同一套接口，保证两处状态一致。
   * 仅查当前这一个供应商，避免拉全量关注列表。
   */
  const supplierId = item?.supplierId ?? null;
  const followedIdsQuery = trpc.search.followedSupplierIds.useQuery(
    { supplierIds: supplierId ? [supplierId] : [] },
    { enabled: isAuthenticated && !!supplierId, retry: false }
  );
  const following =
    !!supplierId && (followedIdsQuery.data ?? []).includes(supplierId);
  const toggleFollowMut = trpc.search.toggleFollowSupplier.useMutation({
    onSuccess: res => {
      followedIdsQuery.refetch();
      toast.success(res.following ? "已关注该供应商" : "已取消关注", {
        description: res.following
          ? "可在搜索页用「☆ 我关注的供应商」快速筛选"
          : undefined,
      });
    },
    onError: e => toast.error(e.message || "操作失败，请重试"),
  });
  const handleToggleFollow = () => {
    if (!isAuthenticated) {
      setLoginOpen(true);
      return;
    }
    if (!supplierId) return;
    toggleFollowMut.mutate({ supplierId, follow: !following });
  };

  const addCartMut = trpc.cart.add.useMutation({
    onSuccess: (res, vars) => {
      utils.cart.count.invalidate();
      utils.cart.list.invalidate();
      if (vars.buyNow) {
        if (!res.checkoutToken) {
          toast.error("结算准备失败，请重试");
          return;
        }
        navigate(
          `/cart?checkout=1&token=${encodeURIComponent(res.checkoutToken)}`
        );
      } else {
        toast.success("已加入购物车", {
          description: `购物车共 ${res.count} 种商品`,
          action: { label: "去结算", onClick: () => navigate("/cart") },
        });
      }
    },
    onError: e => toast.error(e.message),
  });

  const handleAddCart = (buyNow = false) => {
    if (!item) return;
    if (inventoryQuantityDisplay(item).spotQuantity === null) return;
    if (!isAuthenticated) {
      toast.info("请先登录", {
        description: buyNow ? "登录后即可在线下单" : "登录后即可加入购物车",
      });
      setLoginOpen(true);
      return;
    }
    addCartMut.mutate({ inventoryId: item.id, qty, buyNow });
  };

  /** 聊一聊：跳转会话并携带询价意图 */
  const handleInquiry = () => {
    if (!item) return;
    if (!isAuthenticated) {
      toast.info("请先登录", { description: "登录后即可向供应商询价" });
      setLoginOpen(true);
      return;
    }
    if (!item.supplierId) {
      toast.info("无法发起询价", { description: "该供应商信息不完整" });
      return;
    }
    /*
     * 新开标签页而不是 window.location.href 整页跳转。
     * 买家往往在详情页比完参数才询价，询价后还要回来看参数或下单，
     * 把详情页替掉会迫使他重新搜一遍。
     */
    launchInquiry({
      userId: item.supplierId,
      inventoryId: item.id,
      buyerUserId: user?.id,
      partNumber: item.partNumber,
      qty: quantityTouched ? qty || moq : null,
      brand: item.brand,
      pkg: item.pkg,
      source: "detail",
    }, target => openInNewTab(target, navigate));
  };

  const chgQty = (d: number) => {
    setQuantityTouched(true);
    setQty(q => Math.max(moq, (q || 0) + d * step));
  };
  const inputQty = (v: string) => {
    setQuantityTouched(true);
    const n = parseInt(v.replace(/\D/g, ""), 10);
    if (isNaN(n) || n < moq) {
      toast(`最低起订量为 ${fmt(moq)} 颗`);
      setQty(moq);
    } else {
      setQty(n);
    }
  };

  /* 基础属性表 */
  const baseAttrs: Array<[string, string]> = item
    ? ([
        ["型号", item.partNumber],
        ["品牌/厂牌", item.brand],
        ["分类", item.category || "—"],
        ["封装", item.pkg || "—"],
        ["生产周期（Date Code）", item.dateCode || "—"],
        ["货期", displayedLeadTime],
        ["RoHS", item.rohs ? "符合" : "—"],
        ["整包数量 SPQ", item.spq ? `${fmt(item.spq)} 颗` : "—"],
        ["增值税发票", item.invoice ? "可开 13% 专票" : "不提供"],
      ] as Array<[string, string]>)
    : [];

  const notFound = isError && error?.data?.code === "NOT_FOUND";

  return (
    <div className="pd-root">
      <MallNav />
      <MallLoginDialog open={loginOpen} onOpenChange={setLoginOpen} />
      {inquiryEnterpriseChooser}

      {/* ── 面包屑 ── */}
      <div className="pd-crumb">
        <Link href="/">首页</Link>
        <span className="pd-crumb-sep">/</span>
        <Link href="/search">元器件搜索</Link>
        <span className="pd-crumb-sep">/</span>
        <span className="pd-crumb-cur">
          {item?.partNumber || previewDetail?.partNumber || "商品详情"}
        </span>
      </div>

      {!validId || notFound ? (
        <div className="pd-empty">
          <div className="pd-empty-title">商品不存在或已下架</div>
          <div className="pd-empty-sub">该库存可能已售罄或被供应商下架</div>
          <button className="pd-btn-back" onClick={() => navigate("/search")}>
            去搜索其他元器件
          </button>
        </div>
      ) : isLoading ? (
        <ProductDetailSkeleton />
      ) : !isAuthenticated && previewDetail ? (
        <section className="pd-public-preview" aria-label="商品安全概览">
          <div className="pd-public-preview-copy">
            <span className="pd-public-preview-eyebrow">商品安全概览</span>
            <h1>{previewDetail.partNumber}</h1>
            {previewDetail.brand && <p>品牌：{previewDetail.brand}</p>}
            <span
              className={`pd-public-availability${previewDetail.availability === "in_stock" ? " is-in-stock" : ""}`}
            >
              {previewAvailabilityLabel(previewDetail.availability)}
            </span>
          </div>
          <div className="pd-public-preview-cta">
            <p>登录后可查看报价、阶梯价、库存明细及供应商信息。</p>
            <button type="button" onClick={() => setLoginOpen(true)}>
              {PUBLIC_QUOTE_CTA}
            </button>
          </div>
        </section>
      ) : isError ? (
        <div className="pd-empty">加载失败，请刷新页面重试</div>
      ) : item ? (
        <>
          {/* ── 主区 ── */}
          <div
            ref={mainExposureRef}
            className="pd-main"
            data-material-part={item.partNumber}
          >
            {/* 图片区 */}
            <ProductImageGallery
              sources={photos}
              partNumber={item.partNumber}
            />

            {/* 信息区 */}
            <div className="pd-info">
              <h1 className="pd-part">{item.partNumber}</h1>
              <div className="pd-summary">
                {[item.brand, item.pkg, item.category]
                  .filter(Boolean)
                  .join(" · ")}
              </div>
              <div className="pd-attrs">
                <div className="pd-attr">
                  <span className="pl">厂牌</span>
                  <span className="pv">{item.brand}</span>
                </div>
                <div className="pd-attr">
                  <span className="pl">封装</span>
                  <span className="pv">{item.pkg || "—"}</span>
                </div>
                <div className="pd-attr">
                  <span className="pl">生产周期（Date Code）</span>
                  <span className="pv">{item.dateCode || "—"}</span>
                </div>
                <div className="pd-attr">
                  <span className="pl">货期</span>
                  <span
                    className="pv"
                    style={{
                      color: isSpotInventoryLeadTime(item.leadTime)
                        ? "#1e7a44"
                        : "#c9610d",
                    }}
                  >
                    {displayedLeadTime}
                  </span>
                </div>
                <div className="pd-attr">
                  <span className="pl">起订量</span>
                  <span className="pv">{fmt(moq)} 颗</span>
                </div>
                <div className="pd-attr">
                  <span className="pl">库存</span>
                  <span className="pv pd-stock">
                    {formatInventoryAvailability(item) || "—"}
                  </span>
                </div>
              </div>
              <div className="pd-supplier">
                <span className="pd-sup-label">供应商</span>
                <span className="pd-sup-name">
                  {item.supplierName || SUPPLIER_NAME_FALLBACK}
                </span>
                {/* 星标关注：就地完成，不离开详情页 */}
                {supplierId && (
                  <button
                    type="button"
                    className={`pd-sup-follow${following ? " is-following" : ""}`}
                    title={following ? "取消关注该供应商" : "关注该供应商"}
                    aria-label={following ? "取消关注该供应商" : "关注该供应商"}
                    aria-pressed={following}
                    onClick={handleToggleFollow}
                  >
                    {following ? "★ 已关注" : "☆ 关注"}
                  </button>
                )}
                <span className="pd-sup-badge">实名认证</span>
              </div>
              {item.note && <div className="pd-note">备注：{item.note}</div>}
            </div>

            {/* 购买区 */}
            <div className="pd-buy">
              <div className="pd-price-box">
                <div className="pd-price-title">
                  {negotiable
                    ? "价格"
                    : untaxed
                      ? hasRealLadder(item.ladderPrices)
                        ? "阶梯报价（未税价）"
                        : "未税价"
                      : hasRealLadder(item.ladderPrices)
                        ? "阶梯报价（含税）"
                        : "价格（含税）"}
                </div>
                {negotiable ? (
                  <div className="pd-negotiable">
                    <span className="pd-neg-tag">面议</span>
                    <span className="pd-neg-hint">价格请与供应商洽谈</span>
                  </div>
                ) : hasRealLadder(item.ladderPrices) ? (
                  ladder.map((t, i) => (
                    <div
                      key={i}
                      className={`pd-lp-row${i === ti ? " active" : ""}`}
                    >
                      <span className="pd-lp-qty">{fmt(t.minQty)}+</span>
                      <span className="pd-lp-price">
                        ¥{formatInventoryUnitPrice(t.price)}
                      </span>
                    </div>
                  ))
                ) : (
                  <div className="pd-lp-row active">
                    <span className="pd-lp-qty">{fmt(moq)}+</span>
                    <span className="pd-lp-price">
                      ¥{formatInventoryUnitPrice(ladder[0]?.price ?? 0)}
                    </span>
                  </div>
                )}
              </div>

              {!negotiable && !untaxed && canTransact && (
                <>
                  <div className="pd-qty-row">
                    <span className="pd-qty-label">数量</span>
                    <div className="pd-stepper">
                      <button onClick={() => chgQty(-1)}>−</button>
                      <input
                        type="text"
                        value={qty}
                        onChange={e => {
                          setQuantityTouched(true);
                          setQty(
                            parseInt(e.target.value.replace(/\D/g, ""), 10) || 0
                          );
                        }}
                        onBlur={e => inputQty(e.target.value)}
                      />
                      <button onClick={() => chgQty(1)}>＋</button>
                    </div>
                    <span className="pd-qty-unit">颗</span>
                  </div>
                  <div className="pd-subtotal">
                    单价 ¥{formatInventoryUnitPrice(unitPrice)} · 小计{" "}
                    <b>¥{fmt(subtotal)}</b>
                  </div>
                </>
              )}

              <div className="pd-actions">
                {!negotiable && !untaxed && canTransact && (
                  <>
                    <button
                      className="pd-btn-cart"
                      onClick={() => handleAddCart(false)}
                      disabled={addCartMut.isPending}
                    >
                      加入购物车
                    </button>
                    <button
                      className="pd-btn-buy"
                      onClick={() => handleAddCart(true)}
                      disabled={addCartMut.isPending}
                    >
                      立即购买
                    </button>
                  </>
                )}
                <button
                  className={`pd-btn-inquiry${negotiable || untaxed ? " primary" : ""}`}
                  onClick={handleInquiry}
                >
                  <span className="pd-cdot" />
                  聊一聊
                </button>
              </div>
              {untaxed && (
                <div className="pd-buy-tips">
                  该价格不含税费，最终含税价格、发票、运费及交货方式请通过「聊一聊」确认
                </div>
              )}
            </div>
          </div>

          {/* ── 详细参数 ── */}
          <div className="pd-section">
            <div className="pd-sec-title">规格参数</div>
            <div className="pd-spec-table">
              {baseAttrs.map(([k, v]) => (
                <div key={k} className="pd-spec-row">
                  <span className="pd-spec-key">{k}</span>
                  <span className="pd-spec-val">{v}</span>
                </div>
              ))}
              {specsLoading && specEntries.length === 0 && (
                <div className="pd-spec-row">
                  <span className="pd-spec-key">技术参数</span>
                  <span className="pd-spec-val">参数加载中…</span>
                </div>
              )}
              {specEntries.map(([k, v]) => (
                <div key={`dict-${k}`} className="pd-spec-row">
                  <span className="pd-spec-key">{k}</span>
                  <span className="pd-spec-val">{v}</span>
                </div>
              ))}
            </div>
            {specs?.datasheetUrl && (
              <a
                className="pd-datasheet"
                href={specs.datasheetUrl}
                target="_blank"
                rel="noopener noreferrer"
              >
                查看数据手册 ↗
              </a>
            )}
          </div>

          {/* ── 相关商品推荐：独立状态，不阻塞当前库存 ── */}
          {relatedLoading && related.length === 0 && (
            <div className="pd-section">
              <div className="pd-sec-title">相关商品推荐</div>
              <div className="pd-secondary-loading">推荐加载中…</div>
            </div>
          )}
          {relatedError && related.length === 0 && (
            <div className="pd-section">
              <div className="pd-sec-title">相关商品推荐</div>
              <div className="pd-secondary-loading">
                推荐暂不可用，不影响当前商品查看与询价
              </div>
            </div>
          )}
          {related.length > 0 && (
            <div className="pd-section">
              <div className="pd-sec-title">相关商品推荐</div>
              <div ref={relatedExposureRef} className="pd-related">
                {related.map(r => {
                  const rPrice = Number(r.priceEx ?? 0);
                  return (
                    <button
                      key={r.id}
                      className="pd-rel-card"
                      data-material-part={r.partNumber}
                      onClick={() => openInNewTab(`/product/${r.id}`, navigate)}
                    >
                      <ProductImageGallery
                        compact
                        sources={productImageSources(r)}
                        partNumber={r.partNumber}
                      />
                      <div className="pd-rel-pn">{r.partNumber}</div>
                      <div className="pd-rel-meta">
                        {[r.brand, r.pkg].filter(Boolean).join(" · ")}
