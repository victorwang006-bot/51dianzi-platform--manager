import * as React from "react";
import type { OfficialMaterialResource } from "@shared/officialMaterial";
import { safeOfficialDatasheetUrl } from "@/lib/officialMaterialClient";

type Parameter = OfficialMaterialResource["parameters"][number];
const PRIORITIES = [
  /^(?:cpu)?内核$|处理器内核|^core$|cpu core/i,
  /主频|时钟频率|工作频率|^frequency$|clock|speed/i,
  /程序存储|闪存|flash|rom size|program memory/i,
  /ram|sram|运行内存|数据存储/i,
  /通道|channels?/i,
  /带宽|bandwidth/i,
  /失调|offset/i,
  /分辨率|resolution/i,
  /工作电压|供电电压|supply voltage/i,
  /温度|temperature/i,
  /^(can|i2c|spi|usb|usart|uart|io|i\/o)/i,
];
/** Reject only whole-field placeholders; zero, signed numbers and ranges are real values. */
export function isValidOfficialParameterText(value: string): boolean {
  const text = value.trim();
  return text.length > 0 && !/^(?:[-—–]\s*)+$/.test(text)
    && !/^(?:n\/?a|n\.a\.|null|undefined|none|暂无(?:数据|参数|资料)?|待确认|未知|无数据|未提供|未填写|未设置|不详)$/i.test(text);
}
export function validOfficialParameters(parameters: readonly Parameter[]): Parameter[] {
  return parameters.filter(row => isValidOfficialParameterText(row.name) && isValidOfficialParameterText(row.value));
}
/** Rank real dictionary keys only; never infer specifications from a part number. */
export function selectOfficialCoreParameters(parameters: readonly Parameter[], limit = 5): Parameter[] {
  return validOfficialParameters(parameters)
    .map((row, index) => ({ row, index, rank: PRIORITIES.findIndex(pattern => pattern.test(row.name)) }))
    .sort((a, b) => (a.rank < 0 ? PRIORITIES.length : a.rank) - (b.rank < 0 ? PRIORITIES.length : b.rank) || a.index - b.index)
    .slice(0, Math.min(5, Math.max(0, limit))).map(entry => entry.row);
}
export type OfficialMaterialResourcesProps = {
  publicId: string;
  partNumber: string;
  resource?: OfficialMaterialResource;
  favorite?: boolean;
  authenticated: boolean;
  pending: boolean;
  error?: string | null;
  resourceError?: string | null;
  onFavorite: () => void;
  onResourceRetry?: () => void;
  /** Search-only parameter panel; omitted/default preserves existing shared consumers. */
  presentation?: "default" | "search";
};
export type OfficialMaterialResourcesDependencies = {
  React: Pick<typeof React, "useState" | "useEffect" | "useRef" | "createElement" | "Fragment">;
};
/** Card-local disclosure: no overlay, focus trap, provider or independent query client. */
export function createOfficialMaterialResources(deps: OfficialMaterialResourcesDependencies) {
  const React = deps.React;
  return function OfficialMaterialResources({ publicId, partNumber, resource, favorite, authenticated, pending, error, resourceError, onFavorite, onResourceRetry, presentation = "default" }: OfficialMaterialResourcesProps) {
    const [open, setOpen] = React.useState(false);
    const parameters = validOfficialParameters(resource?.parameters ?? []);
    const core = selectOfficialCoreParameters(parameters);
    const extra = parameters.filter(row => !core.includes(row));
    const expanded = open && extra.length > 0;
    const extraId = `omr-extra-parameters-${publicId}`;
    const technicalError = resourceError === undefined ? error : resourceError;
    const resourceUnavailable = Boolean(technicalError);
    const pdf = safeOfficialDatasheetUrl(resource?.datasheetUrl);
    const searchPresentation = presentation === "search";
    React.useEffect(() => {
      setOpen(false);
    }, [publicId]);
    const summary = core.length > 0 ? <dl className="omr-core-parameters" aria-label="核心参数">
      {core.map((row, index) => <div key={`${row.name}:${index}`}><dt>{row.name}</dt><dd title={row.value}>{row.value}</dd></div>)}
    </dl> : <p className="omr-parameter-state" role="status">
      {resourceUnavailable ? technicalError : resource ? "暂无参数" : "参数加载中"}
    </p>;
    const disclosure = extra.length > 0 && <button type="button" className="omr-more-parameters" aria-label={expanded ? `收起 ${partNumber} 额外参数` : `展开 ${partNumber} 额外参数`}
      aria-expanded={expanded} aria-controls={extraId} title={expanded ? "收起额外参数" : `还有${extra.length}项参数`}
      onClick={() => setOpen(value => !value)}>{searchPresentation ? expanded ? "收起参数 ∧" : "更多参数 ∨" : expanded ? "收起 ▴" : "…"}</button>;
    const extras = extra.length > 0 && <dl id={extraId} className="omr-extra-parameters" aria-label={`${partNumber} 额外参数`} hidden={!expanded}>
      {expanded && extra.map((row, index) => <div key={`${row.name}:${index}`}><dt>{row.name}</dt><dd>{row.value}</dd></div>)}
    </dl>;
    return <>
      {!searchPresentation && summary}
      <div className="omr-material-utilities">
        <button type="button" className={favorite ? "is-favorite" : undefined} aria-pressed={favorite === true}
          disabled={pending || (authenticated && favorite === undefined)} onClick={onFavorite}
          title={authenticated && favorite === undefined ? "收藏状态加载中" : undefined}>
          <svg width="13" height="13" viewBox="0 0 24 24" aria-hidden="true" fill={favorite ? "currentColor" : "none"} stroke="currentColor" strokeWidth="1.6"><path d="m12 3 2.8 5.7 6.2.9-4.5 4.4 1.1 6.2-5.6-3-5.6 3 1.1-6.2L3 9.6l6.2-.9Z" /></svg>
          {pending ? "处理中" : favorite ? "已收藏" : "收藏"}
        </button>
        {resourceUnavailable && onResourceRetry && <button type="button" onClick={onResourceRetry}>重试资料</button>}
        {pdf && <a href={pdf} target="_blank" rel="noopener noreferrer">PDF规格书</a>}
        {!searchPresentation && disclosure}
        {error && (core.length > 0 || error !== technicalError) && <span className="omr-material-error" role="status">{error}</span>}
      </div>
      {searchPresentation ? <section className="omr-parameter-panel" data-omr-zone="parameters" aria-label={`${partNumber} 商品参数`}>
        {summary}
        {extras}
        {disclosure}
      </section> : extras}
    </>;
  };
}
export default createOfficialMaterialResources({ React });
